cmake_minimum_required(VERSION 3.16)
project(MonicaRecomp C RC)
set(SDL3_DIR "${CMAKE_CURRENT_SOURCE_DIR}/work/deps/sdl-build" CACHE PATH "SDL3 CMake package")
find_package(SDL3 CONFIG REQUIRED)
set(CMAKE_C_STANDARD 11)
set(ENGINE "${CMAKE_CURRENT_LIST_DIR}/work/deps/smsggrecomp" CACHE PATH "Runtime experimental")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/runtime/glue.c")
  set(MONICA_RUNTIME "${CMAKE_CURRENT_SOURCE_DIR}/runtime")
else()
  set(MONICA_RUNTIME "${ENGINE}/runner")
endif()
file(GLOB BANK_SOURCES CONFIGURE_DEPENDS generated/monica_bank*.c)
# Large conservative switches compile much faster at O1. Keep the scheduler,
# device runtime and UI at the normal Release optimization level.
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  set_source_files_properties(${BANK_SOURCES} PROPERTIES COMPILE_OPTIONS "-O1")
endif()
add_library(monica-core STATIC ${MONICA_RUNTIME}/glue.c src/replay.c
  ${ENGINE}/runner/video/sms_vdp.c ${ENGINE}/runner/audio/sn76489.c
  ${ENGINE}/runner/external/superzazu/z80.c
  generated/monica_full.c generated/monica_dispatch.c generated/monica_layout.c
  generated/monica_step.c ${BANK_SOURCES})
target_include_directories(monica-core PUBLIC ${MONICA_RUNTIME} src ${ENGINE}/runner ${ENGINE}/runner/include generated)
if(NOT MSVC)
  target_link_libraries(monica-core PUBLIC m)
endif()
add_executable(monica-headless ${ENGINE}/runner/main.c)
target_link_libraries(monica-headless PRIVATE monica-core)
add_executable(monica-decoder-verify tests/decoder_halt.c ${ENGINE}/recompiler/src/z80_decoder.c ${ENGINE}/runner/external/superzazu/z80.c)
target_include_directories(monica-decoder-verify PRIVATE ${ENGINE}/recompiler/src ${ENGINE}/runner/external/superzazu)
# These assertions are the regression contract, including in Release builds.
if(NOT MSVC)
  target_compile_options(monica-decoder-verify PRIVATE -UNDEBUG)
else()
  target_compile_options(monica-decoder-verify PRIVATE /UNDEBUG)
endif()
add_executable(monica-verify src/verify.c)
target_link_libraries(monica-verify PRIVATE monica-core)
add_executable(MonicaRecomp WIN32 src/app.c src/theme.c src/input.c src/preferences.c src/choice.c src/app.rc)
set_target_properties(MonicaRecomp PROPERTIES OUTPUT_NAME "MonicaRecomp-v0.10")
add_executable(monica-theme-preview src/theme_preview.c src/theme.c)
target_include_directories(monica-theme-preview PRIVATE ${ENGINE}/runner)
target_link_libraries(monica-theme-preview PRIVATE user32 gdi32 ole32 windowscodecs uuid m msimg32)
target_compile_definitions(MonicaRecomp PRIVATE UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_link_libraries(MonicaRecomp PRIVATE monica-core SDL3::SDL3 user32 gdi32 comdlg32 shell32 winmm ole32 windowscodecs uuid comctl32 uxtheme msimg32)
add_executable(monica-system-verify src/system_verify.c src/input.c src/preferences.c src/choice.c)
add_executable(monica-interface-preview src/interface_preview.c src/input.c src/preferences.c src/choice.c)
target_include_directories(monica-interface-preview PRIVATE ${ENGINE}/runner)
target_compile_definitions(monica-interface-preview PRIVATE UNICODE _UNICODE)
target_link_libraries(monica-interface-preview PRIVATE SDL3::SDL3 user32 gdi32 shell32 winmm ole32 windowscodecs uuid comctl32 uxtheme m)
target_compile_definitions(monica-system-verify PRIVATE UNICODE _UNICODE)
target_link_libraries(monica-system-verify PRIVATE SDL3::SDL3 user32 gdi32 shell32 winmm ole32 windowscodecs uuid comctl32 uxtheme msimg32)
add_custom_command(TARGET MonicaRecomp POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:MonicaRecomp>/assets"
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:MonicaRecomp>/assets"
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/LICENSES" "$<TARGET_FILE_DIR:MonicaRecomp>/LICENSES"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE" "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE_SCOPE.md" "${CMAKE_CURRENT_SOURCE_DIR}/THIRD_PARTY.md" "$<TARGET_FILE_DIR:MonicaRecomp>"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:SDL3::SDL3-shared>" "$<TARGET_FILE_DIR:MonicaRecomp>")
if(MINGW)
  target_link_options(monica-verify PRIVATE -municode)
  target_link_options(MonicaRecomp PRIVATE -municode)
  target_link_options(monica-theme-preview PRIVATE -municode)
  target_link_options(monica-system-verify PRIVATE -municode)
  target_link_options(monica-interface-preview PRIVATE -municode)
endif()
